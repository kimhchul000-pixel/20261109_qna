📌 웹에서 시트로 바로 기록하기 (Apps Script 웹 앱 URL)

구글 시트 상단 [확장 프로그램] → [Apps Script]에 아래 코드를 붙여넣고 [배포] → [새 배포] (웹 앱, 모든 사용자)로 배포한 뒤 나오는 URL을 아래에 입력해 주세요. 

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var data = JSON.parse(e.postData.contents);
    if (data.action === 'ask') {
      var now = new Date();
      var timeStr = data.time || Utilities.formatDate(now, "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");
      
      // 시트 1행 헤더(질문, 답변, id, 시간) 열 위치를 자동 감지합니다
      var headers = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 4)).getValues()[0];
      var colQ = -1, colA = -1, colId = -1, colTime = -1;
      for (var i = 0; i < headers.length; i++) {
        var h = String(headers[i] || '').trim().toLowerCase();
        if (h.indexOf('질문') !== -1 || h === 'question') colQ = i;
        else if (h.indexOf('답변') !== -1 || h === 'answer') colA = i;
        else if (h === 'id' || h.indexOf('작성자') !== -1 || h.indexOf('사용자') !== -1) colId = i;
        else if (h.indexOf('시간') !== -1 || h.indexOf('time') !== -1 || h.indexOf('일시') !== -1) colTime = i;
      }
      // 헤더 미지정 시 기본 순서: A(질문), B(답변), C(id), D(시간)
      if (colQ === -1) colQ = 0;
      if (colA === -1) colA = 1;
      if (colId === -1) colId = 2;
      if (colTime === -1) colTime = 3;
      
      var maxCol = Math.max(colQ, colA, colId, colTime) + 1;
      var rowData = new Array(maxCol).fill('');
      rowData[colQ] = data.question || '';
      rowData[colA] = ''; // 강의자가 시트 B열에서 직접 작성
      rowData[colId] = data.userId || '익명';
      rowData[colTime] = timeStr;
      
      sheet.appendRow(rowData);
      return ContentService.createTextOutput(JSON.stringify({ ok: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } finally {
    lock.releaseLock();
  }
}
